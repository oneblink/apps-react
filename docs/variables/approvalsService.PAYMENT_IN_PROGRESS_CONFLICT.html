    Variable PAYMENT_IN_PROGRESS_CONFLICTConst

    PAYMENT_IN_PROGRESS_CONFLICT: "PAYMENT_IN_PROGRESS" = 'PAYMENT_IN_PROGRESS'

    Conflict code on the error thrown when an approval step already has a checkout in progress.

    Example

    import {
      approvalsService,
      OneBlinkAppsError,
    } from '@oneblink/apps-react'

    try {
      await approvalsService.createApprovalStepPayment(
        formSubmissionApprovalId,
        {
          redirectUrl,
          paymentFormUrl,
          form,
          submissionId,
        },
      )
    } catch (error) {
      if (
        error instanceof OneBlinkAppsError &&
        error.conflict === approvalsService.PAYMENT_IN_PROGRESS_CONFLICT
      ) {
        console.log('A payment is already in progress')
      }
    }
